1. Anchor - 锚点
================================================================================

AnchorProps:
  - className?: string - 自定义类名
  - style?: CSSProperties - 自定义样式
  - children?: ReactNode - 子元素
  - offsetTop?: number - 距离顶部偏移量
  - affix?: boolean - 是否固定
  - bounds?: number - 边界值
  - getContainer: () => HTMLElement - 获取容器函数
  - onChange?: (activeLink: string) => void - 锚点切换回调

AnchorLinkProps:
  - href: string - 锚点链接
  - title: string - 锚点标题
  - children?: ReactNode - 子元素
  - className?: string - 自定义类名
  - style?: CSSProperties - 自定义样式

2. Breadcrumb - 面包屑
================================================================================

BreadcrumbProps:
  - items: BreadcrumbItem[] - 面包屑项数组
  - separator?: string - 分隔符
  - className?: string - 自定义类名

BreadcrumbItem:
  - label: ReactNode - 显示文本
  - href?: string - 链接地址

3. Button - 按钮
================================================================================

ButtonProps:
  - children?: ReactNode - 子元素/按钮文本
  - variant?: 'primary' | 'secondary' | 'danger' | 'success' | 'warning' | 'link' - 按钮变体
  - type?: 'primary' | 'secondary' | 'danger' | 'success' | 'warning' | 'link' | 'button' | 'submit' | 'reset' - 按钮类型（已废弃，请使用variant）
  - size?: 'small' | 'medium' | 'large' - 按钮大小
  - disabled?: boolean - 是否禁用
  - loading?: boolean - 是否加载中
  - onClick?: () => void - 点击回调
  - className?: string - 自定义类名
  - style?: CSSProperties - 自定义样式
  - icon?: string | ReactNode - 图标
  - htmlType?: 'button' | 'submit' | 'reset' - HTML按钮类型
  - href?: string - 链接地址（当variant为link时有效）
  - title?: string - 按钮标题

5. Carousel - 轮播图
================================================================================

CarouselProps:
  - items: CarouselItem[] - 轮播数据
  - currentIndex?: number - 当前激活索引（受控）
  - defaultCurrentIndex?: number - 默认激活索引
  - onChange?: (index: number) => void - 切换回调
  - autoplay?: boolean - 是否自动播放
  - interval?: number - 自动播放间隔(ms)
  - effect?: CarouselEffect - 切换动画效果
  - direction?: CarouselDirection - 切换方向
  - showIndicators?: boolean - 是否显示指示器
  - indicatorPosition?: CarouselIndicatorPosition - 指示器位置
  - showArrows?: boolean - 是否显示箭头
  - loop?: boolean - 是否循环播放
  - pauseOnHover?: boolean - 悬停时是否暂停
  - className?: string - 自定义类名
  - style?: CSSProperties - 自定义样式
  - duration?: number - 切换动画时长(ms)
  - indicatorProgress?: boolean - 是否启用指示器进度动画

CarouselItem:
  - key: string - 唯一标识
  - image: string - 图片地址
  - title?: ReactNode - 标题
  - description?: ReactNode - 描述
  - link?: string - 点击跳转链接
  - render?: () => ReactNode - 自定义渲染内容
  - infoPosition?: CarouselInfoPosition - 文案位置

CarouselEffect: 'slide' | 'fade' | 'flip' | 'cards' | 'creative' | 'coverflow' | 'parallax' | 'zoom' | 'book' | 'curtain' | 'mosaic' | 'rain'
CarouselDirection: 'horizontal' | 'vertical'
CarouselIndicatorPosition: 'bottom' | 'top' | 'left' | 'right' | 'center'
CarouselArrowPosition: 'inside' | 'outside' | 'none'
CarouselInfoPosition: 'bottom' | 'top' | 'left' | 'right' | 'center'

================================================================================

CascaderProps:

  - placeholder?: string - 占位符
  - disabled?: boolean - 是否禁用
  - size?: 'large' | 'middle' | 'small' - 大小
  - style?: CSSProperties - 自定义样式
  - className?: string - 自定义类名
  - allowClear?: boolean - 是否允许清除
  - maxTagCount?: number - 最大标签数量
  - showSearch?: boolean - 是否支持搜索
  - expandTrigger?: 'click' | 'hover' - 展开触发方式
  - changeOnSelect?: boolean - 是否选中即改变
  - fieldNames?: { label?: string; value?: string; children?: string } - 自定义字段名
  - width?: number | string - 宽度
  - dropdownWidth?: number | string - 下拉框宽度
  - dropdownHeight?: number | string - 下拉框高度
  - dropdownStyle?: CSSProperties - 下拉框样式
  - dropdownClassName?: string - 下拉框类名
  - placement?: 'bottomLeft' | 'bottomRight' | 'topLeft' | 'topRight' - 弹出位置
  - checkbox?: boolean - 是否显示复选框
  - autoWidth?: boolean - 是否自动宽度

CascaderOption:
  - value: any - 选项值
  - label: ReactNode - 选项标签
  - disabled?: boolean - 是否禁用
  - children?: CascaderOption[] - 子选项

  - name?: string - 字段名

  - className?: string - 自定义类名
  - style?: CSSProperties - 自定义样式
  - help?: ReactNode - 帮助信息

  - colon?: boolean - 是否显示冒号
  - labelSpan?: number - 标签占位格数
  - wrapperSpan?: number - 输入框占位格数

  - children?: ReactNode - 子元素

Rule:

  - message?: string - 错误提示信息
  - pattern?: RegExp - 正则表达式
  - min?: number - 最小值
  - max?: number - 最大值
  - len?: number - 长度
  - type?: string - 类型
  - validator?: (rule: Rule, value: any) => Promise<void> | void - 自定义校验函数
  - whitespace?: boolean - 是否不允许空白字符

FormInstance:

  - setFieldValue: (name: string, value: any) => void - 设置字段值

  - setFields: (fields: { name: string; errors?: string[]; value?: any }[]) => void - 设置字段状态
  - resetFields: (names?: string[]) => void - 重置字段
  - validateFields: (names?: string[]) => Promise<Record<string, any>> - 校验字段
  - submit: () => void - 提交表单
  - destroy: () => void - 销毁表单

Hooks:
  - useForm(): [FormInstance] - 创建表单实例
